Research Question

Do roads designed for peak capacity contribute to catastrophic crash outcomes during off-peak hours?

Hypothesis: Roads designed for peak traffic volumes have excess capacity during off-peak periods, enabling higher speeds and more severe crashes.

# Helper function to read spatial data from Box
read_box_sf <- function(file_id, crs = 2272) {
  temp_dir <- tempdir()
  dl_file_path <- box_dl(file_id, local_dir = temp_dir, overwrite = TRUE)
  sf_data <- st_read(dl_file_path, quiet = TRUE) %>% st_transform(crs)
  return(sf_data)
}

philly_crashes <- philly_crashes %>%
  mutate(
    # Hour of crash
time_raw = case_when(
  !is.na(DISPATCH_TM) ~ as.character(DISPATCH_TM),
  !is.na(TIME_OF_DAY) ~ as.character(TIME_OF_DAY),
  TRUE ~ NA_character_
),

time_clean = str_pad(time_raw, width = 4, side = "left", pad = "0"),

HOUR_OF_DAY_CLEAN = as.integer(str_sub(time_clean, 1, 2)),

HOUR_OF_DAY_CLEAN = case_when(
  HOUR_OF_DAY_CLEAN >= 0 & HOUR_OF_DAY_CLEAN <= 23 ~ HOUR_OF_DAY_CLEAN,
  TRUE ~ NA_integer_
),
    
    # KSI flag (Killed or Suspected Serious Injury)
    is_ksi = if_else(FATAL_OR_SUSP_SERIOUS_INJ == "1", 1, 0, missing = 0),
    
    # Time periods (peak: 7-10 AM, 4-7 PM)
    time_period = case_when(
      is.na(HOUR_OF_DAY_CLEAN) ~ "Unknown",
      HOUR_OF_DAY_CLEAN >= 7 & HOUR_OF_DAY_CLEAN < 10 ~ "AM Peak (7-10)",
      HOUR_OF_DAY_CLEAN >= 16 & HOUR_OF_DAY_CLEAN < 19 ~ "PM Peak (16-19)",
      HOUR_OF_DAY_CLEAN >= 10 & HOUR_OF_DAY_CLEAN < 16 ~ "Midday (10-16)",
      HOUR_OF_DAY_CLEAN >= 19 & HOUR_OF_DAY_CLEAN < 22 ~ "Evening (19-22)",
      HOUR_OF_DAY_CLEAN >= 22 | HOUR_OF_DAY_CLEAN < 7 ~ "Night/Late (22-7)",
      TRUE ~ "Other"
    ),
    
    is_peak = if_else(
      time_period %in% c("AM Peak (7-10)", "PM Peak (16-19)"), 
      1, 0, missing = 0
    ),
    
    # Day of week
    day_name = case_when(
      DAY_OF_WEEK == 1 ~ "Sunday",
      DAY_OF_WEEK == 2 ~ "Monday",
      DAY_OF_WEEK == 3 ~ "Tuesday",
      DAY_OF_WEEK == 4 ~ "Wednesday",
      DAY_OF_WEEK == 5 ~ "Thursday",
      DAY_OF_WEEK == 6 ~ "Friday",
      DAY_OF_WEEK == 7 ~ "Saturday",
      TRUE ~ NA_character_
    ),
    is_weekend = if_else(DAY_OF_WEEK %in% c(1, 7), 1, 0, missing = 0),
    
    # Severity labels
    severity_label = case_when(
      MAX_SEVERITY_LEVEL == "1" ~ "Fatal",
      MAX_SEVERITY_LEVEL == "2" ~ "Suspected Serious Injury",
      MAX_SEVERITY_LEVEL == "3" ~ "Suspected Minor Injury",
      MAX_SEVERITY_LEVEL == "4" ~ "Possible Injury",
      MAX_SEVERITY_LEVEL == "0" ~ "PDO",
      TRUE ~ "Unknown"
    ),
    
    # Convert flags to integer
    PEDESTRIAN = as.integer(PEDESTRIAN),
    BICYCLE = as.integer(BICYCLE),
    VULNERABLE_ROAD_USER = as.integer(VULNERABLE_ROAD_USER),
    SPEEDING_RELATED = as.integer(SPEEDING_RELATED),
    ILLUMINATION_DARK = as.integer(ILLUMINATION_DARK),
    
    # Road design variables
    speed_limit = as.integer(SPEED_LIMIT),
    lane_count = case_when(
      LANE_COUNT == "99" ~ NA_integer_,
      TRUE ~ as.integer(LANE_COUNT)
    ),
    
    # Coordinate validity
    has_valid_coords = !is.na(DEC_LATITUDE) & !is.na(DEC_LONGITUDE),
    
    # Clean street name for matching
    crash_street_clean = str_to_upper(str_trim(STREET_NAME))
  ) %>%
  
  # Set factor levels
  mutate(
    time_period = factor(
      time_period,
      levels = c("AM Peak (7-10)", "Midday (10-16)", "PM Peak (16-19)",
                 "Evening (19-22)", "Night/Late (22-7)", "Unknown")
    ),
    
    severity_label = factor(
      severity_label,
      levels = c("Fatal", "Suspected Serious Injury", "Suspected Minor Injury",
                 "Possible Injury", "PDO", "Unknown")
    )
  )

Part 3: Road Network Linkage

Two-Tier Approach: (1) Street name matching (high confidence), (2) Spatial snap within 100 feet (fallback)

Show code 
# Load Street Centerline shapefile
street_centerline_id <- "2137970615238"

street_roads <- read_box_sf(street_centerline_id, crs = 2272) %>%
  select(seg_id, st_name, st_type, class, responsibl, geometry) %>%
  mutate(
    full_street_name = str_trim(paste(st_name, st_type)),
    full_street_name_clean = str_to_upper(full_street_name)
  )

street_lookup <- street_roads %>% 
  st_drop_geometry() %>%
  distinct(full_street_name_clean, seg_id, st_name, st_type)

# Tier 1: Street name matching
# Strategy: For each crash, keep only the FIRST matching segment to avoid memory explosion
crashes_street_matched <- philly_crashes %>%
  filter(!is.na(crash_street_clean)) %>%
  inner_join(
    street_lookup %>% 
      group_by(full_street_name_clean) %>% 
      slice(1) %>%  # Keep only first segment per street name
      ungroup(),
    by = c("crash_street_clean" = "full_street_name_clean")
  ) %>%
  mutate(road_source = "Street_Name", match_confidence = "High")

# Verify no duplicates

# Tier 2: Spatial snap (for unmatched crashes with coordinates)
crashes_needing_snap <- philly_crashes %>%
  filter(!(CRN %in% crashes_street_matched$CRN), has_valid_coords) %>%
  st_as_sf(coords = c("DEC_LONGITUDE", "DEC_LATITUDE"), crs = 4326) %>%
  st_transform(crs = 2272)

crashes_snapped <- crashes_needing_snap %>%
  mutate(
    nearest_road_id = st_nearest_feature(geometry, street_roads),
    snap_distance = as.numeric(st_distance(geometry, street_roads[nearest_road_id,], by_element = TRUE))
  ) %>%
  filter(snap_distance <= 100) %>%
  mutate(
    road_source = "Spatial_Snap",
    match_confidence = case_when(
      snap_distance <= 50 ~ "High",
      snap_distance <= 75 ~ "Medium",
      TRUE ~ "Low"
    )
  )

# Combine and add to philly_crashes
tier1_df <- crashes_street_matched %>%
  select(CRN, seg_id, road_source, match_confidence)

tier2_df <- crashes_snapped %>%
  mutate(seg_id = street_roads$seg_id[nearest_road_id]) %>%
  st_drop_geometry() %>%
  select(CRN, seg_id, road_source, match_confidence, snap_distance)

# Combine tier1 and tier2 - ensure no duplicates
combined_linkage <- bind_rows(tier1_df, tier2_df) %>%
  distinct(CRN, .keep_all = TRUE)
